cmake_minimum_required(VERSION 3.16)
project(wanren CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")

add_library(wanren_core STATIC core/params.cpp core/formation.cpp core/sim.cpp core/ai.cpp core/scenario.cpp)
target_include_directories(wanren_core PUBLIC core)
find_package(Threads REQUIRED)
target_link_libraries(wanren_core PUBLIC Threads::Threads)

add_executable(wanren_bench bench/bench.cpp)
target_link_libraries(wanren_bench wanren_core)
add_executable(wanren_test tests/test_sim.cpp)
target_link_libraries(wanren_test wanren_core)

# 图形程序：SDL2 + OpenGL 3.3；文本 macOS 用 CoreText，其他平台用 SDL2_ttf
find_package(SDL2 QUIET)
find_package(OpenGL QUIET)
if(SDL2_FOUND AND OpenGL_FOUND)
  if(APPLE)
    enable_language(OBJCXX)
    add_executable(wanren app/main.cpp app/gfx.cpp app/text_mac.mm)
    target_compile_definitions(wanren PRIVATE GL_SILENCE_DEPRECATION)
    target_link_libraries(wanren wanren_core SDL2::SDL2 OpenGL::GL "-framework CoreText" "-framework CoreGraphics" "-framework CoreFoundation")
  else()
    find_package(SDL2_ttf REQUIRED)
    add_executable(wanren app/main.cpp app/gfx.cpp app/text_sdl.cpp)
    target_link_libraries(wanren wanren_core SDL2::SDL2 OpenGL::GL SDL2_ttf::SDL2_ttf)
  endif()
else()
  message(STATUS "未找到 SDL2/OpenGL，只编译核心、基准与测试")
endif()
